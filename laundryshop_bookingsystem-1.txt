    // =========================================================
    // REGISTER FUNCTION
    // =========================================================

    private void register() {
        String full_name =
            nameField.getText().trim();

    String email_name =
            emailField.getText().trim();

    String user_name =
            registerUsername.getText().trim();

    String password =
            new String(
                    registerPassword.getPassword()
            );

    String confirm =
            new String(
                    confirmPassword.getPassword()
            );

    // =====================================================
    // VALIDATION 1 - EMPTY FIELDS
    // =====================================================

    if (
            full_name.isEmpty()
            ||
            email_name.isEmpty()
            ||
            user_name.isEmpty()
            ||
            password.isEmpty()
            ||
            confirm.isEmpty()
    ) {
        JOptionPane.showMessageDialog(
                this,
                "Please fill in all fields.",
                "Registration Error",
                JOptionPane.ERROR_MESSAGE
        );

        return;
    }

    // =====================================================
    // VALIDATION 2 - EMAIL
    // =====================================================

    if (
            !email_name.contains("@")
            ||
            !email_name.contains(".")
    ) {
        JOptionPane.showMessageDialog(
                this,
                "Please enter a valid email address.",
                "Registration Error",
                JOptionPane.ERROR_MESSAGE
        );

        return;
    }

    // =====================================================
    // VALIDATION 3 - PASSWORD LENGTH
    // =====================================================

    if (password.length() < 6) {
        JOptionPane.showMessageDialog(
                this,
                "Password must be at least 6 characters.",
                "Registration Error",
                JOptionPane.ERROR_MESSAGE
        );

        return;
    }

    // =====================================================
    // VALIDATION 4 - CONFIRM PASSWORD
    // =====================================================

    if (!password.equals(confirm)) {
        JOptionPane.showMessageDialog(
                this,
                "Passwords do not match.",
                "Registration Error",
                JOptionPane.ERROR_MESSAGE
        );

        return;
    }

    // =====================================================
    // SAVE TO MYSQL
    // =====================================================

    String sql =
            "INSERT INTO users "
            + "(full_name, email, username, password) "
            + "VALUES (?, ?, ?, ?)";

    try (
            Connection conn = DBConnection.connect();
            PreparedStatement pst =
                    conn.prepareStatement(sql)
    ) {
        pst.setString(1, full_name);
        pst.setString(2, email_name);
        pst.setString(3, user_name);
        pst.setString(4, password);

        int result =
                pst.executeUpdate();

        if (result > 0) {
            JOptionPane.showMessageDialog(
                    this,
                    "Registration successful!\n"
                    + "Your account has been saved.",
                    "Success",
                    JOptionPane.INFORMATION_MESSAGE
            );

            // Clear register fields

            nameField.setText("");
            emailField.setText("");
            registerUsername.setText("");
            registerPassword.setText("");
            confirmPassword.setText("");

            // Clear login fields

            loginUsername.setText("");
            loginPassword.setText("");

            // Go back to Login

            cardLayout.show(
                    cards,
                    "LOGIN"
            );
        }

    } catch (SQLException e) {
        if (
                e.getMessage()
                        .toLowerCase()
                        .contains("duplicate")
        ) {
            JOptionPane.showMessageDialog(
                    this,
                    "Username or email already exists.",
                    "Registration Error",
                    JOptionPane.ERROR_MESSAGE
            );

        } else {
            JOptionPane.showMessageDialog(
                    this,
                    "Database error:\n"
                    + e.getMessage(),
                    "Database Error",
                    JOptionPane.ERROR_MESSAGE
            );

            e.printStackTrace();
        }
    }
}

    // =========================================================
    // LOGIN FUNCTION
    // =========================================================

    private void login() {
        String username =
            loginUsername
                    .getText()
                    .trim();

    String password =
            new String(
                    loginPassword.getPassword()
            );

    // =====================================================
    // CHECK EMPTY FIELDS
    // =====================================================

    if (
            username.isEmpty()
            ||
            password.isEmpty()
    ) {
        JOptionPane.showMessageDialog(
                this,
                "Please enter your username/email and password.",
                "Login Error",
                JOptionPane.ERROR_MESSAGE
        );

        return;
    }

    // =====================================================
    // SEARCH DATABASE
    // =====================================================

    String sql =
            "SELECT full_name, username, password, role, is_archived "
            + "FROM users "
            + "WHERE username = ? OR email = ?";

    try (
            Connection conn = DBConnection.connect();
            PreparedStatement pst =
                    conn.prepareStatement(sql)
    ) {
        pst.setString(1, username);
        pst.setString(2, username);

        ResultSet rs =
                pst.executeQuery();

        if (rs.next()) {
            String databaseName =
                    rs.getString("full_name");

            String databasePassword =
                    rs.getString("password");

            String role = rs.getString("role");

            if (role == null || role.trim().isEmpty()) {
                role = "Customer";
            }

            boolean archived = rs.getBoolean("is_archived");

            // =================================================
            // CHECK PASSWORD
            // =================================================

            if (
                    password.equals(
                            databasePassword
                    )
            ) {
                // Archived accounts (e.g. resigned staff) cannot log in
                if (archived) {
                    JOptionPane.showMessageDialog(
                            this,
                            "This account has been archived.\n"
                            + "Please contact the administrator.",
                            "Account Archived",
                            JOptionPane.WARNING_MESSAGE
                    );

                    return;
                }

                JOptionPane.showMessageDialog(
                        this,
                        "Welcome, "
                        + databaseName
                        + "!",
                        "Login Successful",
                        JOptionPane.INFORMATION_MESSAGE
                );

                // =================================================
                // OPEN HOME PAGE
                // =================================================

                if (role.equalsIgnoreCase("Admin")
                        || role.equalsIgnoreCase("Staff")) {

                    // Staff / Admin -> manage all bookings
                    new AdminBookingsPage(databaseName, role)
                            .setVisible(true);

                } else {

                    // Customer -> normal home page
                    new HomePagee(databaseName).setVisible(true);
                }

                // Close login window

                dispose();

            } else {
                JOptionPane.showMessageDialog(
                        this,
                        "Incorrect password.",
                        "Login Error",
                        JOptionPane.ERROR_MESSAGE
                );
            }

        } else {
            JOptionPane.showMessageDialog(
                    this,
                    "Account not found.",
                    "Login Error",
                    JOptionPane.ERROR_MESSAGE
            );
        }

    } catch (SQLException e) {
        JOptionPane.showMessageDialog(
                this,
                "Database error:\n"
                + e.getMessage(),
                "Database Error",
                JOptionPane.ERROR_MESSAGE
        );

        e.printStackTrace();
    }
}

    // =========================================================
    // CIRCULAR LAVANDERA LOGO
    // =========================================================

    static class CircularBadgePanel
            extends JPanel {
        @Override
        protected void paintComponent(
                Graphics g
        ) {
            super.paintComponent(g);

            Graphics2D g2 =
                    (Graphics2D) g.create();

            // =================================================
            // ANTI-ALIASING
            // =================================================

            g2.setRenderingHint(
                    RenderingHints.KEY_ANTIALIASING,
                    RenderingHints.VALUE_ANTIALIAS_ON
            );

            int w = getWidth();

            int h = getHeight();

            // =================================================
            // OUTER CIRCLE
            // =================================================

            g2.setColor(
                    new Color(
                            190,
                            160,
                            215
                    )
            );

            g2.fill(
                    new Ellipse2D.Float(
                            0,
                            0,
                            w,
                            h
                    )
            );

            // =================================================
            // INNER CIRCLE
            // =================================================

            int inset = 6;

            g2.setColor(
                    new Color(
                            180,
                            145,
                            205
                    )
            );

            g2.fill(
                    new Ellipse2D.Float(
                            inset,
                            inset,
                            w - inset * 2,
                            h - inset * 2
                    )
            );

            // =================================================
            // WHITE BORDER
            // =================================================

            g2.setStroke(
                    new BasicStroke(2f)
            );

            g2.setColor(
                    Color.WHITE
            );

            g2.draw(
                    new Ellipse2D.Float(
                            inset,
                            inset,
                            w - inset * 2,
                            h - inset * 2
                    )
            );

            // =================================================
            // DROPLET
            // =================================================

            g2.setColor(
                    Color.WHITE
            );

            g2.setStroke(
                    new BasicStroke(2.2f)
            );

            Path2D drop =
                    new Path2D.Float();

            float cx =
                    w / 2f;

            float topY =
                    h * 0.22f;

            drop.moveTo(
                    cx,
                    topY
            );

            drop.curveTo(
                    cx - w * 0.11f,
                    topY + h * 0.16f,

                    cx - w * 0.09f,
                    topY + h * 0.30f,

                    cx,
                    topY + h * 0.30f
            );

            drop.curveTo(
                    cx + w * 0.09f,
                    topY + h * 0.30f,

                    cx + w * 0.11f,
                    topY + h * 0.16f,

                    cx,
                    topY
            );

            drop.closePath();

            g2.draw(drop);

            // =================================================
            // LAVANDERA TEXT
            // =================================================

            g2.setFont(
                    new Font(
                            "Serif",
                            Font.BOLD,
                            (int) (h * 0.13)
                    )
            );

            FontMetrics fm =
                    g2.getFontMetrics();

            String word =
                    "Lavandera";

            int textW =
                    fm.stringWidth(word);

            g2.drawString(
                    word,
                    cx - textW / 2f,
                    h * 0.58f
            );

            // =================================================
            // LAUNDRY SHOP TEXT
            // =================================================

            g2.setFont(
                    new Font(
                            "SansSerif",
                            Font.PLAIN,
                            (int) (h * 0.065)
                    )
            );

            fm =
                    g2.getFontMetrics();

            String sub =
                    "Laundry Shop";

            int subW =
                    fm.stringWidth(sub);

            g2.drawString(
                    sub,
                    cx - subW / 2f,
                    h * 0.67f
            );

            // =================================================
            // SMALL TEAL DROPLET
            // =================================================

            g2.setColor(
                    new Color(
                            90,
                            200,
                            190
                    )
            );

            g2.fill(
                    new Ellipse2D.Float(
                            cx - w * 0.14f,
                            h * 0.74f,
                            w * 0.07f,
                            w * 0.09f
                    )
            );

            g2.dispose();
        }

        @Override
        public Dimension getPreferredSize() {
            return new Dimension(
                    240,
                    240
            );
        }
    }

    // GRADIENT BACKGROUND
    // =========================================================

    static class GradientBackgroundPanel
            extends JPanel {
        @Override
        protected void paintComponent(
                Graphics g
        ) {
            super.paintComponent(g);

            Graphics2D g2 =
                    (Graphics2D) g.create();

            // =================================================
            // GRADIENT
            // =================================================

            GradientPaint gp =
                    new GradientPaint(

                            0,
                            0,

                            new Color(
                                    240,
                                    220,
                                    235
                            ),

                            getWidth(),
                            getHeight(),

                            new Color(
                                    210,
                                    225,
                                    245
                            )
                    );

            g2.setPaint(gp);

            g2.fillRect(
                    0,
                    0,
                    getWidth(),
                    getHeight()
            );

            g2.dispose();
        }
    }
