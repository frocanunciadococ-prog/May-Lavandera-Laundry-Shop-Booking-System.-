    private JButton addMenuButton(
            JPanel parent,
            String text,
            int y,
            boolean selected,
            Runnable action
    ) {
        JButton button = new JButton(text);
        button.setBounds(15, y, 195, 42);
        button.setFont(new Font("Arial", Font.BOLD, 14));
        button.setForeground(Color.WHITE);
        button.setBackground(selected ? SELECTED_MENU : PURPLE);
        button.setBorderPainted(false);
        button.setFocusPainted(false);
        button.setCursor(new Cursor(Cursor.HAND_CURSOR));
        button.setHorizontalAlignment(SwingConstants.LEFT);
        button.setBorder(new EmptyBorder(0, 20, 0, 0));

        if (selected) {
            selectedMenu = button;
        }
        menuButtons.add(button);

        button.addMouseListener(new MouseAdapter() {
            @Override
            public void mouseEntered(MouseEvent e) {
                button.setBackground(new Color(135, 90, 170));
            }

            @Override
            public void mouseExited(MouseEvent e) {
                button.setBackground(button == selectedMenu
                        ? SELECTED_MENU : PURPLE);
            }
        });

        button.addActionListener(e -> action.run());
        parent.add(button);
        return button;
    }

    /** Highlights the sidebar button of the page that is open. */
    private void selectMenu(JButton active) {
        selectedMenu = active;

        for (JButton b : menuButtons) {
            b.setBackground(b == active ? SELECTED_MENU : PURPLE);
        }
    }

    // ================= LOAD BOOKINGS =================
    private void loadBookings() {
        cardsContainer.removeAll();

        String search = searchField == null
                ? "" : searchField.getText().trim();

        String sql =
                "SELECT booking_id, username, service, booking_date, "
                + "booking_time, booking_status, payment_status, "
                + "quantity, weight, amount "
                + "FROM tbl_bookings "
                + "WHERE CAST(booking_id AS CHAR) LIKE ? "
                + "OR username LIKE ? "
                + "OR service LIKE ? "
                + "ORDER BY booking_id DESC";

        int count = 0;

        try (Connection con = requireConnection();
             PreparedStatement pst = con.prepareStatement(sql)) {

            String keyword = "%" + search + "%";
            pst.setString(1, keyword);
            pst.setString(2, keyword);
            pst.setString(3, keyword);

            try (ResultSet rs = pst.executeQuery()) {
                while (rs.next()) {
                    cardsContainer.add(createBookingCard(
                            rs.getInt("booking_id"),
                            safe(displayName(rs.getString("username"))),
                            safe(rs.getString("service")),
                            safe(rs.getString("booking_date")),
                            safe(rs.getString("booking_time")),
                            safe(rs.getString("booking_status")),
                            safe(rs.getString("payment_status")),
                            rs.getString("quantity"),
                            rs.getDouble("weight"),
                            rs.getDouble("amount")
                    ));

                    cardsContainer.add(Box.createVerticalStrut(10));
                    count++;
                }
            }

            if (count == 0) {
                JLabel emptyLabel = new JLabel(
                        "No booking records found.",
                        SwingConstants.CENTER
                );
                emptyLabel.setFont(new Font("Arial", Font.PLAIN, 16));
                emptyLabel.setForeground(GRAY);
                emptyLabel.setAlignmentX(Component.CENTER_ALIGNMENT);
                cardsContainer.add(Box.createVerticalStrut(40));
                cardsContainer.add(emptyLabel);
            }

        } catch (SQLException e) {
            showDatabaseError(e);
        }

        cardsContainer.revalidate();
        cardsContainer.repaint();
    }

    private JPanel createBookingCard(
            int bookingId,
            String username,
            String service,
            String date,
            String time,
            String bookingStatus,
            String paymentStatus,
            String quantity,
            double weight,
            double amount
    ) {
        // Labels are created first; the card positions them itself
        // (see doLayout below) so it works at ANY window width.
        final JLabel idLabel = new JLabel("Booking #" + bookingId);
        idLabel.setFont(new Font("Arial", Font.BOLD, 16));
        idLabel.setForeground(PURPLE);

        final JLabel customerLabel = new JLabel("Customer: " + username);
        customerLabel.setFont(new Font("Arial", Font.PLAIN, 13));
        customerLabel.setForeground(TEXT);

        final JLabel serviceLabel = new JLabel("Service: " + service);
        serviceLabel.setFont(new Font("Arial", Font.PLAIN, 13));
        serviceLabel.setForeground(TEXT);

        final JLabel dateLabel = new JLabel(
                "Date: " + date + "  |  Time: " + time);
        dateLabel.setFont(new Font("Arial", Font.PLAIN, 13));
        dateLabel.setForeground(TEXT);

        final JLabel amountLabel = new JLabel(
                "Amount: \u20B1" + String.format("%,.2f", amount)
        );
        amountLabel.setFont(new Font("Arial", Font.BOLD, 14));
        amountLabel.setForeground(PURPLE);

        final JLabel bookingLabel = new JLabel("Booking: " + bookingStatus);
        bookingLabel.setFont(new Font("Arial", Font.PLAIN, 13));
        bookingLabel.setForeground(TEXT);

        final JLabel paymentLabel = new JLabel("Payment: " + paymentStatus);
        paymentLabel.setFont(new Font("Arial", Font.BOLD, 13));
        paymentLabel.setForeground(
                PAYMENT_PAID.equalsIgnoreCase(paymentStatus) ? GREEN : RED
        );

        // Right-hand column: two action slots (top / bottom).
        final JComponent[] action = new JComponent[2];
        final boolean cancelled =
                STATUS_CANCELLED.equalsIgnoreCase(bookingStatus);

        // A booking cancelled by the customer has nothing left to pay
        // or finish, so the action buttons are replaced by a label.
        if (cancelled) {
            JLabel cancelledLabel = new JLabel(
                    "Cancelled by customer", SwingConstants.CENTER
            );
            cancelledLabel.setFont(new Font("Arial", Font.BOLD, 13));
            cancelledLabel.setForeground(RED);
            action[0] = cancelledLabel;

        } else {
            // Only staff/admin users should be able to change a booking.
            boolean actionsAllowed = canManagePayments();

            JButton paymentButton = new JButton(
                    PAYMENT_PAID.equalsIgnoreCase(paymentStatus)
                    ? "Unpaid" : " Paid"
            );
            styleButton(paymentButton);
            paymentButton.setEnabled(actionsAllowed);

            paymentButton.addActionListener(e -> {
                String newStatus = PAYMENT_PAID.equalsIgnoreCase(paymentStatus)
                        ? PAYMENT_NOT_PAID : PAYMENT_PAID;

                int confirm = JOptionPane.showConfirmDialog(
                        this,
                        "Change payment status to " + newStatus + "?",
                        "Confirm Payment Update",
                        JOptionPane.YES_NO_OPTION
                );

                if (confirm == JOptionPane.YES_OPTION) {
                    updatePayment(bookingId, newStatus);
                }
            });

            action[0] = paymentButton;

            if (STATUS_DONE.equalsIgnoreCase(bookingStatus)) {
                // "Dialog" font has the check mark; Arial shows a box.
                JLabel doneLabel = new JLabel(
                        "\u2713 Done", SwingConstants.CENTER
                );
                doneLabel.setFont(new Font("Dialog", Font.BOLD, 14));
                doneLabel.setForeground(GREEN);
                action[1] = doneLabel;

            } else {
                JButton doneButton = new JButton("Done");
                styleButton(doneButton);
                doneButton.setBackground(GREEN);
                doneButton.setEnabled(actionsAllowed);

                doneButton.addActionListener(e -> {
                    int confirm = JOptionPane.showConfirmDialog(
                            this,
                            "Mark Booking #" + bookingId + " as done?\n"
                            + "The customer will be notified that their "
                            + "laundry is finished.",
                            "Confirm Done",
                            JOptionPane.YES_NO_OPTION
                    );

                    if (confirm == JOptionPane.YES_OPTION) {
                        markDone(bookingId);
                    }
                });

                action[1] = doneButton;
            }
        }

        // "Create Receipt" sits under the payment line. A cancelled
        // booking has nothing to pay, so it gets no receipt.
        final RoundButton[] receiptButton = new RoundButton[1];

        if (!cancelled) {
            RoundButton receipt = new RoundButton("Create Receipt",
                    VERY_LIGHT_PURPLE, LIGHT_PURPLE, PURPLE, LIGHT_PURPLE);
            receipt.setFont(new Font("Arial", Font.BOLD, 11));

            receipt.addActionListener(e -> showReceipt(new ReportRow(
                    bookingId, username, service, date, time,
                    bookingStatus, paymentStatus,
                    quantity == null ? "" : quantity,
                    weight, amount)));

            receiptButton[0] = receipt;
        }

        // The card re-positions everything from its own width, so the
        // action buttons stay pinned to the right edge and nothing is cut
        // off when the window is smaller than maximized.
        JPanel card = new RoundPanel(Color.WHITE, 16) {
            @Override
            public void doLayout() {
                int w = getWidth();

                int actionW = 155;
                int actionX = Math.max(0, w - actionW - 20);
                int midX = Math.max(260, actionX - 195);
                int leftW = Math.max(100, midX - 25);

                idLabel.setBounds(15, 10, leftW, 25);
                customerLabel.setBounds(15, 38, leftW, 22);
                serviceLabel.setBounds(15, 62, leftW, 22);
                dateLabel.setBounds(15, 86, leftW, 22);

                amountLabel.setBounds(midX, 12, 180, 25);
                bookingLabel.setBounds(midX, 43, 180, 22);
                paymentLabel.setBounds(midX, 68, 180, 22);

                if (receiptButton[0] != null) {
                    receiptButton[0].setBounds(midX, 98, 125, 26);
                }

                if (cancelled) {
                    action[0].setBounds(actionX - 10, 48, actionW + 20, 35);
                } else {
                    action[0].setBounds(actionX, 25, actionW, 35);
                    action[1].setBounds(actionX, 75, actionW, 35);
                }
            }
        };

        card.setLayout(null);
        card.setPreferredSize(new Dimension(865, 135));
        card.setMaximumSize(new Dimension(Integer.MAX_VALUE, 135));
        card.setMinimumSize(new Dimension(300, 135));

        card.add(idLabel);
        card.add(customerLabel);
        card.add(serviceLabel);
        card.add(dateLabel);
        card.add(amountLabel);
        card.add(bookingLabel);
        card.add(paymentLabel);

        for (JComponent c : action) {
            if (c != null) {
                card.add(c);
            }
        }

        if (receiptButton[0] != null) {
            card.add(receiptButton[0]);
        }

        return card;
    }

    // ================= MARK DONE + NOTIFY CUSTOMER =================
    private void markDone(int bookingId) {
        if (!guardActiveAccount()) {
            return;
        }

        if (!canManagePayments()) {
            JOptionPane.showMessageDialog(
                    this,
                    "You are not allowed to update this booking.",
                    "Access Denied",
                    JOptionPane.WARNING_MESSAGE
            );
            return;
        }

        try (Connection con = requireConnection()) {

            CustomerNotifications.ensureTable(con);

            String customer;
            String service;

            try (PreparedStatement pst = con.prepareStatement(
                    "SELECT username, service, booking_status "
                    + "FROM tbl_bookings WHERE booking_id = ?")) {

                pst.setInt(1, bookingId);

                try (ResultSet rs = pst.executeQuery()) {
                    if (!rs.next()) {
                        JOptionPane.showMessageDialog(
                                this,
                                "Booking record was not found."
                        );
                        return;
                    }

                    customer = rs.getString("username");
                    service = safe(rs.getString("service"));

                    if (STATUS_CANCELLED.equalsIgnoreCase(
                            rs.getString("booking_status"))) {
                        JOptionPane.showMessageDialog(
                                this,
                                "This booking was cancelled by the customer."
                        );
                        loadBookings();
                        updateSummary();
                        return;
                    }
                }
            }

            int updated;

            try (PreparedStatement pst = con.prepareStatement(
                    "UPDATE tbl_bookings SET booking_status = ? "
                    + "WHERE booking_id = ? "
                    + "AND LOWER(booking_status) <> 'cancelled'")) {

                pst.setString(1, STATUS_DONE);
                pst.setInt(2, bookingId);
                updated = pst.executeUpdate();
            }

            if (updated > 0) {
                CustomerNotifications.add(
                        con,
                        customer,
                        bookingId,
                        "Good news! Your laundry (Booking #" + bookingId
                        + " - " + service + ") is done and ready."
                );

                JOptionPane.showMessageDialog(
                        this,
                        "Booking marked as done.\n"
                        + "The customer has been notified."
                );
            }

            loadBookings();
            updateSummary();

        } catch (SQLException e) {
            showDatabaseError(e);
        }
    }

    // ================= UPDATE PAYMENT =================
    private void updatePayment(int bookingId, String newStatus) {
        if (!guardActiveAccount()) {
            return;
        }

        if (!canManagePayments()) {
            JOptionPane.showMessageDialog(
                    this,
                    "You are not allowed to update payment status.",
                    "Access Denied",
                    JOptionPane.WARNING_MESSAGE
            );
            return;
        }

        String sql = "UPDATE tbl_bookings SET payment_status = ? "
                + "WHERE booking_id = ? "
                + "AND LOWER(booking_status) <> 'cancelled'";

        try (Connection con = requireConnection();
             PreparedStatement pst = con.prepareStatement(sql)) {

            pst.setString(1, newStatus);
            pst.setInt(2, bookingId);

            int updated = pst.executeUpdate();

            if (updated > 0) {
                JOptionPane.showMessageDialog(
                        this,
                        "Payment status updated successfully."
                );
                loadBookings();
                updateSummary();
            } else {
                JOptionPane.showMessageDialog(
                        this,
                        "Booking record was not found."
                );
            }

        } catch (SQLException e) {
            showDatabaseError(e);
        }
    }

    // ================= SUMMARY COUNTS =================
    private void updateSummary() {
        String sql = "SELECT "
                + "COUNT(*) AS total_bookings, "
                + "COALESCE(SUM(CASE "
                + "WHEN LOWER(booking_status) = 'pending' "
                + "THEN 1 ELSE 0 END), 0) AS pending_bookings, "
                + "COALESCE(SUM(CASE "
                + "WHEN LOWER(payment_status) = 'paid' "
                + "THEN 1 ELSE 0 END), 0) AS paid_bookings, "
                + "COALESCE(SUM(CASE "
                + "WHEN LOWER(payment_status) = 'not paid' "
                + "THEN 1 ELSE 0 END), 0) AS unpaid_bookings "
                + "FROM tbl_bookings";

        try (Connection con = requireConnection();
             PreparedStatement pst = con.prepareStatement(sql);
             ResultSet rs = pst.executeQuery()) {

            if (rs.next()) {
                totalBookingsLabel.setText(
                        String.valueOf(rs.getInt("total_bookings"))
                );
                pendingBookingsLabel.setText(
                        String.valueOf(rs.getInt("pending_bookings"))
                );
                paidBookingsLabel.setText(
                        String.valueOf(rs.getInt("paid_bookings"))
                );
                unpaidBookingsLabel.setText(
                        String.valueOf(rs.getInt("unpaid_bookings"))
                );
            }

        } catch (SQLException e) {
            showDatabaseError(e);
        }
    }

    /** One booking row used by the reports page. */
    private static class ReportRow {
        final int id;
        final String customer;
        final String service;
        final String date;
        final String time;
        final String status;
        final String payment;
        final String quantity;
        final double weight;
        final double amount;

        ReportRow(int id, String customer, String service, String date,
                  String time, String status, String payment,
                  String quantity, double weight, double amount) {
            this.id = id;
            this.customer = customer;
            this.service = service;
            this.date = date;
            this.time = time;
            this.status = status;
            this.payment = payment;
            this.quantity = quantity;
            this.weight = weight;
            this.amount = amount;
        }

        boolean cancelled() {
            return STATUS_CANCELLED.equalsIgnoreCase(status);
        }

        boolean done() {
            return STATUS_DONE.equalsIgnoreCase(status);
        }

        /** Money actually received: paid and not cancelled. */
        boolean sale() {
            return !cancelled() && PAYMENT_PAID.equalsIgnoreCase(payment);
        }

        boolean unpaid() {
            return !cancelled()
                    && PAYMENT_NOT_PAID.equalsIgnoreCase(payment);
        }

        String statusLabel() {
            if (cancelled()) return "Cancelled";
            if (done()) return "Completed";
            return status.isEmpty() ? "Pending" : status;
        }
    }

    /** Allows only digits and "-" while typing a yyyy-MM-dd date. */
    private static class DateInputFilter extends DocumentFilter {

        private static final int MAX_LENGTH = 10;

        @Override
        public void insertString(
                FilterBypass fb, int offset, String text, AttributeSet attrs
        ) throws BadLocationException {
            replace(fb, offset, 0, text, attrs);
        }

        @Override
        public void replace(
                FilterBypass fb, int offset, int length,
                String text, AttributeSet attrs
        ) throws BadLocationException {

            String current = fb.getDocument().getText(
                    0, fb.getDocument().getLength());

            String replacement = text == null ? "" : text;

            String next = current.substring(0, offset)
                    + replacement
                    + current.substring(offset + length);

            if (next.length() <= MAX_LENGTH && next.matches("[0-9-]*")) {
                super.replace(fb, offset, length, replacement, attrs);
            }
        }
    }

    // ---------- report logic ----------

    /** True while the code (not the user) is changing the date fields. */
    private boolean suppressDateEvents;

    /**
     * Default range = the WHOLE current month (1st to last day).
     *
     * <p>A booking date is the scheduled laundry date, which is often
     * in the future, so the range must not stop at today or upcoming
     * bookings would be hidden from the report.</p>
     */
    private void setDefaultReportDates() {
        boolean before = suppressDateEvents;
        suppressDateEvents = true;

        try {
            LocalDate today = LocalDate.now();
            reportFromField.setText(today.withDayOfMonth(1).toString());
            reportToField.setText(today.with(
                    TemporalAdjusters.lastDayOfMonth()).toString());
        } finally {
            suppressDateEvents = before;
        }
    }

    /** An empty field (no limit) or a complete yyyy-MM-dd date is ready. */
    private static boolean isDateReady(String text) {
        if (text.isEmpty()) {
            return true;
        }

        try {
            LocalDate.parse(text);
            return true;
        } catch (DateTimeParseException e) {
            return false;
        }
    }

    /**
     * Refreshes the report after the user typed or picked a date.
     * Stays silent (no pop-ups) while the date is still half typed.
     */
    private void autoRefreshReport() {
        if (suppressDateEvents
                || reportsPanel == null
                || !reportsPanel.isVisible()
                || reportFromField == null
                || reportToField == null) {
            return;
        }

        String from = reportFromField.getText().trim();
        String to = reportToField.getText().trim();

        if (!isDateReady(from) || !isDateReady(to)) {
            return;
        }

        if (!from.isEmpty() && !to.isEmpty()
                && LocalDate.parse(from).isAfter(LocalDate.parse(to))) {
            return;
        }

        generateReport();
    }

    private void resetReportFilters() {
        setDefaultReportDates();
        reportTypeBox.setSelectedIndex(0);
        reportSearchField.setText("");
        generateReport();
    }

    private void applyQuickDate(String preset) {
        LocalDate today = LocalDate.now();

        suppressDateEvents = true;

        try {
            switch (preset) {
                case "Today":
                    reportFromField.setText(today.toString());
                    reportToField.setText(today.toString());
                    break;
                case "This Week":
                    // Sunday to Saturday, so bookings later this week
                    // (scheduled dates) are included too.
                    reportFromField.setText(today.with(
                            TemporalAdjusters.previousOrSame(DayOfWeek.SUNDAY))
                            .toString());
                    reportToField.setText(today.with(
                            TemporalAdjusters.nextOrSame(DayOfWeek.SATURDAY))
                            .toString());
                    break;
                case "This Month":
                    setDefaultReportDates();
                    break;
                default:
                    reportFromField.setText("");
                    reportToField.setText("");
                    break;
            }
        } finally {
            suppressDateEvents = false;
        }

        generateReport();
    }

    /** Loads the rows for the chosen dates, then refreshes the page. */
    private void generateReport() {
        if (!guardActiveAccount()) {
            return;
        }

        String from = reportFromField.getText().trim();
        String to = reportToField.getText().trim();

        try {
            LocalDate fromDate = from.isEmpty() ? null : LocalDate.parse(from);
            LocalDate toDate = to.isEmpty() ? null : LocalDate.parse(to);

            if (fromDate != null && toDate != null
                    && fromDate.isAfter(toDate)) {
                JOptionPane.showMessageDialog(
                        this,
                        "\"Date From\" cannot be after \"Date To\".",
                        "Invalid Dates",
                        JOptionPane.WARNING_MESSAGE
                );
                return;
            }
        } catch (DateTimeParseException e) {
            JOptionPane.showMessageDialog(
                    this,
                    "Please type dates as yyyy-MM-dd.\n"
                    + "Example: 2026-10-15\n"
                    + "Leave a date empty for no limit.",
                    "Invalid Date",
                    JOptionPane.WARNING_MESSAGE
            );
            return;
        }

        StringBuilder sql = new StringBuilder(
                "SELECT booking_id, username, service, booking_date, "
                + "booking_time, booking_status, payment_status, "
                + "quantity, weight, amount FROM tbl_bookings WHERE 1=1");

        if (!from.isEmpty()) {
            sql.append(" AND booking_date >= ?");
        }
        if (!to.isEmpty()) {
            sql.append(" AND booking_date <= ?");
        }
        sql.append(" ORDER BY booking_date DESC, booking_id DESC");

        reportRows.clear();

        try (Connection con = requireConnection();
             PreparedStatement pst = con.prepareStatement(sql.toString())) {

            int index = 1;
            if (!from.isEmpty()) {
                pst.setString(index++, from);
            }
            if (!to.isEmpty()) {
                pst.setString(index++, to);
            }

            try (ResultSet rs = pst.executeQuery()) {
                while (rs.next()) {
                    reportRows.add(new ReportRow(
                            rs.getInt("booking_id"),
                            safe(displayName(rs.getString("username"))),
                            safe(rs.getString("service")),
                            safe(rs.getString("booking_date")),
                            safe(rs.getString("booking_time")),
                            safe(rs.getString("booking_status")),
                            safe(rs.getString("payment_status")),
                            safe(rs.getString("quantity")),
                            rs.getDouble("weight"),
                            rs.getDouble("amount")
                    ));
                }
            }

        } catch (SQLException e) {
            showDatabaseError(e);
            return;
        }

        if (from.isEmpty() && to.isEmpty()) {
            reportPeriodText = "All Available Historical Data";
        } else {
            reportPeriodText = (from.isEmpty() ? "Start" : from)
                    + " to " + (to.isEmpty() ? "Latest" : to);
        }

        renderReport();
    }

    private boolean matchesReportType(ReportRow r, int typeIndex) {
        switch (typeIndex) {
            case 1:  return r.sale();
            case 2:  return r.done();
            case 3:  return r.cancelled();
            case 4:  return r.unpaid();
            default: return true;
        }
    }

    /** Applies the type + search filters and refreshes all widgets. */
    private void renderReport() {
        if (reportModel == null) {
            return;
        }

        int typeIndex = Math.max(0, reportTypeBox.getSelectedIndex());
        String query = reportSearchField.getText().trim()
                .toLowerCase(Locale.ROOT);

        shownRows.clear();

        for (ReportRow r : reportRows) {
            if (!matchesReportType(r, typeIndex)) {
                continue;
            }

            if (!query.isEmpty()
                    && !r.customer.toLowerCase(Locale.ROOT).contains(query)
                    && !r.service.toLowerCase(Locale.ROOT).contains(query)
                    && !String.valueOf(r.id).contains(query)) {
                continue;
            }

            shownRows.add(r);
        }

        int completed = 0;
        int cancelled = 0;
        int unpaid = 0;
        double sales = 0;

        reportModel.setRowCount(0);

        for (ReportRow r : shownRows) {
            if (r.done()) completed++;
            if (r.cancelled()) cancelled++;
            if (r.unpaid()) unpaid++;
            if (r.sale()) sales += r.amount;

            reportModel.addRow(new Object[]{
                r.id, r.date, r.customer, r.service,
                r.statusLabel(), r.payment, r.amount
            });
        }

        metricSales.setText("\u20B1" + String.format("%,.2f", sales));
        metricBookings.setText(String.valueOf(shownRows.size()));
        metricCompleted.setText(String.valueOf(completed));
        metricCancelled.setText(String.valueOf(cancelled));
        metricUnpaid.setText(String.valueOf(unpaid));
        reportPeriodLabel.setText("Period: " + reportPeriodText);

        donutChart.setData(completed, cancelled, unpaid);
    }

        try {
            return java.time.LocalTime.parse(time).format(

        }

    /**
     * Prints the receipt. Choose "Microsoft Print to PDF" (or
     * "Save as PDF") as the printer to get a PDF file.
     */
    private void printReceipt(JComponent receipt) {
        PrinterJob job = PrinterJob.getPrinterJob();
        job.setJobName("Lavandera Receipt");

        PageFormat format = job.defaultPage();
        format.setOrientation(PageFormat.PORTRAIT);

        job.setPrintable((Graphics g, PageFormat page, int pageIndex) -> {
            if (pageIndex > 0) {
                return Printable.NO_SUCH_PAGE;
            }

            Graphics2D g2 = (Graphics2D) g;
            g2.translate(page.getImageableX(), page.getImageableY());

            double scale = Math.min(1.6, Math.min(
                    page.getImageableWidth() / receipt.getWidth(),
                    page.getImageableHeight() / receipt.getHeight()));

            g2.scale(scale, scale);
            receipt.printAll(g2);

            return Printable.PAGE_EXISTS;
        }, format);

        if (job.printDialog()) {
            try {
                job.print();
            } catch (PrinterException e) {
                JOptionPane.showMessageDialog(
                        this,
                        "Could not print the receipt: " + e.getMessage(),
                        "Print Error",
                        JOptionPane.ERROR_MESSAGE
                );
            }
        }
    }

    /** Saves the receipt as a sharp PNG picture. */
    private void saveReceiptImage(JComponent receipt, ReportRow booking) {
        JFileChooser chooser = new JFileChooser();
        chooser.setDialogTitle("Save Receipt Image");
        chooser.setSelectedFile(new File(
                "Lavandera_Receipt_" + String.format("%06d", booking.id)
                + ".png"));

        if (chooser.showSaveDialog(this) != JFileChooser.APPROVE_OPTION) {
            return;
        }

        File file = chooser.getSelectedFile();

        if (!file.getName().toLowerCase(Locale.ROOT).endsWith(".png")) {
            file = new File(file.getParentFile(), file.getName() + ".png");
        }

        int scale = 2; // 2x for a sharper picture

        BufferedImage image = new BufferedImage(
                receipt.getWidth() * scale,
                receipt.getHeight() * scale,
                BufferedImage.TYPE_INT_RGB);

        Graphics2D g2 = image.createGraphics();
        g2.setRenderingHint(RenderingHints.KEY_ANTIALIASING,
                RenderingHints.VALUE_ANTIALIAS_ON);
        g2.setRenderingHint(RenderingHints.KEY_TEXT_ANTIALIASING,
                RenderingHints.VALUE_TEXT_ANTIALIAS_ON);
        g2.setColor(Color.WHITE);
        g2.fillRect(0, 0, image.getWidth(), image.getHeight());
        g2.scale(scale, scale);
        receipt.printAll(g2);
        g2.dispose();

        try {
            ImageIO.write(image, "png", file);

            JOptionPane.showMessageDialog(
                    this,
                    "Receipt saved to:\n" + file.getAbsolutePath(),
                    "Receipt Saved",
                    JOptionPane.INFORMATION_MESSAGE
            );

        } catch (IOException e) {
            JOptionPane.showMessageDialog(
                    this,
                    "Could not save the receipt: " + e.getMessage(),
                    "Save Error",
                    JOptionPane.ERROR_MESSAGE
            );
        }
    }

    private void showProfile() {
        if (!guardActiveAccount()) return;

        String fullName = userName;
        String username = "";
        String email = "N/A";
        String roleText = role;
        String memberSince = "N/A";

        String sql = "SELECT full_name, username, email, role FROM users "
                + "WHERE full_name=? AND LOWER(role)=LOWER(?) "
                + "AND is_archived=0 LIMIT 1";

        try (Connection c = requireConnection();
             PreparedStatement q = c.prepareStatement(sql)) {
            q.setString(1, userName);
            q.setString(2, role);
            try (ResultSet r = q.executeQuery()) {
                if (r.next()) {
                    fullName = safe(r.getString("full_name"));
                    username = safe(r.getString("username"));
                    email = safe(r.getString("email"));
                    roleText = safe(r.getString("role"));
                }
            }
        } catch (SQLException e) {
            showDatabaseError(e);
            return;
        }

        // "Member Since" only works if the users table has a created_at
        // column. If it doesn't, this is skipped and "N/A" is shown.
        try (Connection c = requireConnection();
             PreparedStatement q = c.prepareStatement(
                     "SELECT created_at FROM users WHERE full_name=? "
                     + "AND LOWER(role)=LOWER(?) LIMIT 1")) {
            q.setString(1, userName);
            q.setString(2, role);
            try (ResultSet r = q.executeQuery()) {
                if (r.next() && r.getTimestamp(1) != null) {
                    memberSince = r.getTimestamp(1).toLocalDateTime().format(
                            DateTimeFormatter.ofPattern("MMM yyyy", Locale.ENGLISH));
                }
            }
        } catch (SQLException ignored) {
            // column not present - keep "N/A"
        }

        // Split "First Last" (the last word is the last name).
        String trimmed = fullName.trim();
        int space = trimmed.lastIndexOf(' ');
        String firstName = space > 0 ? trimmed.substring(0, space) : trimmed;
        String lastName = space > 0 ? trimmed.substring(space + 1) : "-";

        profileInitials = initialsOf(fullName);
        profileNameLabel.setText(fullName);
        profileRoleLabel.setText(roleText);
        profileUserLabel.setText(username.isEmpty() ? "" : "@" + username);
        profileFirstName.setText(firstName);
        profileLastName.setText(lastName);
        profileUsernameValue.setText(username.isEmpty() ? "-" : username);
        profileEmail.setText(email);
        profileRoleValue.setText(roleText);
        profileSince.setText(memberSince);

        dashboard.setVisible(false);
        reportsPanel.setVisible(false);
        profilePanel.setVisible(true);
        selectMenu(profileButton);
        profileAvatar.repaint();
    }

    private String initialsOf(String name) {
        String[] parts = safe(name).trim().split("\\s+");
        if (parts.length == 0 || parts[0].isEmpty()) return "?";
        String s = parts[0].substring(0, 1);
        if (parts.length > 1) s += parts[parts.length - 1].substring(0, 1);
        return s.toUpperCase(Locale.ROOT);
    }

    // ================= HELPERS =================
    private String safe(String value) {
        return value == null ? "" : value;
    }

    private String escapeHtml(String value) {
        return safe(value)
                .replace("&", "&amp;")
                .replace("<", "&lt;")
                .replace(">", "&gt;")
                .replace("\"", "&quot;")
                .replace("'", "&#39;");
    }

    /**
     * Staff can be archived while their dashboard is still open. This
     * checks the account is still active; if not, it logs the user out.
     *
     * @return {@code true} if the user may continue
     */
    private boolean guardActiveAccount() {

        if (!isStaff()) {
            return true;   // admins cannot be archived
        }

        boolean active;

        try (Connection con = requireConnection();
             PreparedStatement pst = con.prepareStatement(ACTIVE_ACCOUNT_SQL)) {

            pst.setString(1, userName);
            pst.setString(2, ROLE_STAFF);

            try (ResultSet rs = pst.executeQuery()) {
                active = rs.next() && rs.getInt(1) > 0;
            }

        } catch (SQLException e) {
            showDatabaseError(e);
            return false;
        }

        if (!active) {
            JOptionPane.showMessageDialog(
                    this,
                    "This account has been archived.\nYou have been logged out.",
                    "Account Archived",
                    JOptionPane.WARNING_MESSAGE
            );
            dispose();
            new Login().setVisible(true);
            return false;
        }

        return true;
    }

    /**
     * Opens a database connection or fails with a clear message.
     * {@code DBConnection.connect()} returns null when MySQL is not
     * running, which would otherwise surface as a NullPointerException.
     */
    private Connection requireConnection() throws SQLException {
        Connection connection = DBConnection.connect();

        if (connection == null) {
            throw new SQLException(
                    "Cannot connect to the database. "
                    + "Make sure MySQL is running in XAMPP."
            );
        }

        return connection;
    }

    private void showDatabaseError(SQLException e) {
        JOptionPane.showMessageDialog(
                this,
                "Database error:\n" + e.getMessage(),
                "Database Error",
                JOptionPane.ERROR_MESSAGE
        );
    }

    /** Panel inside a scroll pane that always matches the viewport width. */
    private static class WidthTrackingPanel extends JPanel
            implements Scrollable {

        @Override
        public Dimension getPreferredScrollableViewportSize() {
            return getPreferredSize();
        }

        @Override
        public int getScrollableUnitIncrement(
                Rectangle visibleRect, int orientation, int direction) {
            return 16;
        }

        @Override
        public int getScrollableBlockIncrement(
                Rectangle visibleRect, int orientation, int direction) {
            return orientation == SwingConstants.VERTICAL
                    ? visibleRect.height : visibleRect.width;
        }

        @Override
        public boolean getScrollableTracksViewportWidth() {
            return true;
        }

        @Override
        public boolean getScrollableTracksViewportHeight() {
            return false;
        }
    }

    // ================= ROUNDED PANEL =================
    private static class RoundPanel extends JPanel {
        private final Color backgroundColor;
        private final int radius;

        public RoundPanel(Color backgroundColor, int radius) {
            this.backgroundColor = backgroundColor;
            this.radius = radius;
            setOpaque(false);
        }

        @Override
        protected void paintComponent(Graphics graphics) {
            Graphics2D g2 = (Graphics2D) graphics.create();

            g2.setRenderingHint(
                    RenderingHints.KEY_ANTIALIASING,
                    RenderingHints.VALUE_ANTIALIAS_ON
            );

            g2.setColor(backgroundColor);
            g2.fillRoundRect(
                    0, 0, getWidth(), getHeight(),
                    radius, radius
            );

            g2.dispose();
            super.paintComponent(graphics);
        }
    }

    /** Rounded button with a hover color. */
    private static class RoundButton extends JButton {

        private final Color fill;
        private final Color hoverFill;
        private final Color border;
        private boolean hover;

        RoundButton(String text, Color fill, Color hoverFill,
                    Color textColor, Color border) {
            super(text);
            this.fill = fill;
            this.hoverFill = hoverFill;
            this.border = border;

            setForeground(textColor);
            setFont(new Font("Arial", Font.BOLD, 12));
            setContentAreaFilled(false);
            setBorderPainted(false);
            setFocusPainted(false);
            setOpaque(false);
            setMargin(new Insets(0, 4, 0, 4));
            setCursor(new Cursor(Cursor.HAND_CURSOR));

            addMouseListener(new MouseAdapter() {
                @Override
                public void mouseEntered(MouseEvent e) {
                    hover = true;
                    repaint();
                }

                @Override
                public void mouseExited(MouseEvent e) {
                    hover = false;
                    repaint();
                }
            });
        }

        @Override
        protected void paintComponent(Graphics g) {
            Graphics2D g2 = (Graphics2D) g.create();
            g2.setRenderingHint(RenderingHints.KEY_ANTIALIASING,
                    RenderingHints.VALUE_ANTIALIAS_ON);

            g2.setColor(hover ? hoverFill : fill);
            g2.fillRoundRect(0, 0, getWidth() - 1, getHeight() - 1, 12, 12);

            if (border != null) {
                g2.setColor(border);
                g2.drawRoundRect(0, 0, getWidth() - 1, getHeight() - 1,
                        12, 12);
            }

            g2.dispose();
            super.paintComponent(g);
        }
    }

    /** Small rounded square with a symbol in it (metric / header icon). */
    private static class IconBox extends JComponent {

        private final String glyph;
        private final Color fill;
        private final Color fg;
        private final Color border;
        private final int fontSize;

        IconBox(String glyph, Color fill, Color fg, Color border,
                int fontSize) {
            this.glyph = glyph;
            this.fill = fill;
            this.fg = fg;
            this.border = border;
            this.fontSize = fontSize;
        }

        @Override
        protected void paintComponent(Graphics g) {
            Graphics2D g2 = (Graphics2D) g.create();
            g2.setRenderingHint(RenderingHints.KEY_ANTIALIASING,
                    RenderingHints.VALUE_ANTIALIAS_ON);
            g2.setRenderingHint(RenderingHints.KEY_TEXT_ANTIALIASING,
                    RenderingHints.VALUE_TEXT_ANTIALIAS_ON);

            g2.setColor(fill);
            g2.fillRoundRect(0, 0, getWidth() - 1, getHeight() - 1, 10, 10);
            g2.setColor(border);
            g2.drawRoundRect(0, 0, getWidth() - 1, getHeight() - 1, 10, 10);

            g2.setColor(fg);
            g2.setFont(new Font("Dialog", Font.BOLD, fontSize));
            FontMetrics fm = g2.getFontMetrics();
            g2.drawString(glyph,
                    (getWidth() - fm.stringWidth(glyph)) / 2,
                    (getHeight() - fm.getHeight()) / 2 + fm.getAscent());

            g2.dispose();
        }
    }

    /** Text field that shows gray hint text while it is empty. */
    private static class HintField extends JTextField {

        private final String hint;

        HintField(String hint) {
            this.hint = hint;
        }

                g2.drawString(hint, getInsets().left,
                        (getHeight() - fm.getHeight()) / 2 + fm.getAscent());
                g2.dispose();
            }
        }
    }

    /** Doughnut chart of Completed / Cancelled / Unpaid with a legend. */
    private static class DonutChart extends JPanel {

        private static final String[] LABELS = {
            "Completed Orders", "Cancelled Bookings", "Unpaid Orders"
        };

        private static final Color[] COLORS = {GREEN, RED, AMBER};

        private int[] values = {0, 0, 0};

        DonutChart() {
            setOpaque(false);
        }

        void setData(int completed, int cancelled, int unpaid) {
            values = new int[]{completed, cancelled, unpaid};
            repaint();
        }

        @Override
        protected void paintComponent(Graphics g) {
            Graphics2D g2 = (Graphics2D) g.create();
            g2.setRenderingHint(RenderingHints.KEY_ANTIALIASING,
                    RenderingHints.VALUE_ANTIALIAS_ON);
            g2.setRenderingHint(RenderingHints.KEY_TEXT_ANTIALIASING,
                    RenderingHints.VALUE_TEXT_ANTIALIAS_ON);

            int total = values[0] + values[1] + values[2];

            int size = Math.max(20, Math.min(getHeight() - 8, 70));
            int groupWidth = size + 40 + 210;
            int x0 = Math.max(0, (getWidth() - groupWidth) / 2);
            int y0 = (getHeight() - size) / 2;

            float thickness = size * 0.17f;
            double inset = thickness / 2.0;

            g2.setStroke(new BasicStroke(thickness, BasicStroke.CAP_BUTT,
                    BasicStroke.JOIN_MITER));

            if (total == 0) {
                g2.setColor(CARD_BORDER);
                g2.draw(new Ellipse2D.Double(x0 + inset, y0 + inset,
                        size - thickness, size - thickness));
            } else {
                double start = 90;

                for (int i = 0; i < values.length; i++) {
                    if (values[i] <= 0) {
                        continue;
                    }

                    double extent = -360.0 * values[i] / total;

                    g2.setColor(COLORS[i]);
                    g2.draw(new Arc2D.Double(x0 + inset, y0 + inset,
                            size - thickness, size - thickness,
                            start, extent, Arc2D.OPEN));

                    start += extent;
                }
            }

            // legend
            g2.setFont(new Font("Arial", Font.PLAIN, 12));
            FontMetrics fm = g2.getFontMetrics();

            int rowHeight = 20;
            int legendX = x0 + size + 40;
            int legendY = (getHeight() - rowHeight * 3) / 2;

            for (int i = 0; i < LABELS.length; i++) {
                int rowY = legendY + i * rowHeight;

                g2.setColor(COLORS[i]);
                g2.fill(new Ellipse2D.Double(legendX, rowY + 5, 10, 10));

                g2.setColor(TEXT);
                g2.drawString(LABELS[i] + "  (" + values[i] + ")",
                        legendX + 18,
                        rowY + (rowHeight - fm.getHeight()) / 2
                                + fm.getAscent());
            }

            g2.dispose();
        }
    }

    /** Table cell that draws a colored status pill (Completed, Paid...). */
    private static class BadgeRenderer extends JLabel
            implements TableCellRenderer {

        private Color fill = VERY_LIGHT_PURPLE;
        private Color fg = PURPLE;
        private Color border = LIGHT_PURPLE;
        private Color rowBackground = Color.WHITE;
        private String badgeText = "";

        BadgeRenderer() {
            setOpaque(false);
            setFont(new Font("Arial", Font.BOLD, 11));
        }

        @Override
        public Component getTableCellRendererComponent(
                JTable table, Object value, boolean isSelected,
                boolean hasFocus, int row, int column) {

            badgeText = value == null ? "" : value.toString();
            rowBackground = isSelected ? ROW_SELECTED : Color.WHITE;

            String key = badgeText.toLowerCase(Locale.ROOT);

            if (key.equals("completed") || key.equals("paid")) {
                fill = GREEN_SOFT;
                fg = GREEN;
                border = GREEN_BORDER;
            } else if (key.equals("cancelled")) {
                fill = RED_SOFT;
                fg = RED;
                border = RED_BORDER;
            } else if (key.equals("not paid")) {
                fill = AMBER_SOFT;
                fg = AMBER;
                border = AMBER_BORDER;
            } else {
                fill = VERY_LIGHT_PURPLE;
                fg = PURPLE;
                border = LIGHT_PURPLE;
            }

            return this;
        }

        @Override
        protected void paintComponent(Graphics g) {
            Graphics2D g2 = (Graphics2D) g.create();
            g2.setRenderingHint(RenderingHints.KEY_ANTIALIASING,
                    RenderingHints.VALUE_ANTIALIAS_ON);
            g2.setRenderingHint(RenderingHints.KEY_TEXT_ANTIALIASING,
                    RenderingHints.VALUE_TEXT_ANTIALIAS_ON);

            g2.setColor(rowBackground);
            g2.fillRect(0, 0, getWidth(), getHeight());

            g2.setFont(getFont());
            FontMetrics fm = g2.getFontMetrics();

            int pillWidth = fm.stringWidth(badgeText) + 20;
            int pillHeight = 20;
            int x = 12;
            int y = (getHeight() - pillHeight) / 2;

            g2.setColor(fill);
            g2.fillRoundRect(x, y, pillWidth, pillHeight, 10, 10);
            g2.setColor(border);
            g2.drawRoundRect(x, y, pillWidth, pillHeight, 10, 10);

            g2.setColor(fg);
            g2.drawString(badgeText, x + 10,
                    y + (pillHeight - fm.getHeight()) / 2 + fm.getAscent());

            g2.dispose();
        }
    }

    // =====================================================
    // REPORT CALENDAR
    // Unlike the customer booking calendar, reports need
    // PAST dates, so every date can be picked (including
    // Sundays). Has year buttons, Today and Clear.
    // =====================================================

    private static class ReportCalendarDialog extends JDialog {

        private static final String[] WEEKDAYS = {
            "Su", "Mo", "Tu", "We", "Th", "Fr", "Sa"
        };

        // 1 row of weekday names + 6 week rows = 7 rows of 7 cells.
        private static final int GRID_ROWS = 7;

        private final LocalDate currentSelection;

        /** Receives the picked date, or null when "Clear" is pressed. */
        private final Consumer<LocalDate> onPick;

        private YearMonth shownMonth;

        private final JLabel monthLabel =
                new JLabel("", SwingConstants.CENTER);

        private final JPanel dayGrid =
                new JPanel(new GridLayout(GRID_ROWS, 7, 4, 4));

        ReportCalendarDialog(
                Frame owner,
                LocalDate currentSelection,
                Consumer<LocalDate> onPick
        ) {
            super(owner, "Select Date", true);

            this.currentSelection = currentSelection;
            this.onPick = onPick;

            this.shownMonth = YearMonth.from(
                    currentSelection != null
                            ? currentSelection
                            : LocalDate.now());

            JPanel root = new JPanel(new BorderLayout(0, 10));
            root.setBackground(Color.WHITE);
            root.setBorder(BorderFactory.createEmptyBorder(15, 15, 15, 15));

            root.add(buildHeader(), BorderLayout.NORTH);

            dayGrid.setBackground(Color.WHITE);
            dayGrid.setPreferredSize(new Dimension(330, 280));
            root.add(dayGrid, BorderLayout.CENTER);

            root.add(buildFooter(), BorderLayout.SOUTH);

            setContentPane(root);

            renderMonth();

            pack();
            setResizable(false);
            setLocationRelativeTo(owner);
        }

        private JPanel buildHeader() {

            JButton prevYear = navButton("\u00AB", () ->
                    shownMonth = shownMonth.minusYears(1));
            JButton prevMonth = navButton("\u2039", () ->
                    shownMonth = shownMonth.minusMonths(1));
            JButton nextMonth = navButton("\u203A", () ->
                    shownMonth = shownMonth.plusMonths(1));
            JButton nextYear = navButton("\u00BB", () ->
                    shownMonth = shownMonth.plusYears(1));

            monthLabel.setFont(new Font("Arial", Font.BOLD, 15));
            monthLabel.setForeground(TEXT);

            JPanel left = new JPanel(new FlowLayout(FlowLayout.LEFT, 4, 0));
            left.setOpaque(false);
            left.add(prevYear);
            left.add(prevMonth);

            JPanel right = new JPanel(new FlowLayout(FlowLayout.RIGHT, 4, 0));
            right.setOpaque(false);
            right.add(nextMonth);
            right.add(nextYear);

            JPanel header = new JPanel(new BorderLayout());
            header.setBackground(Color.WHITE);
            header.add(left, BorderLayout.WEST);
            header.add(monthLabel, BorderLayout.CENTER);
            header.add(right, BorderLayout.EAST);

            return header;
        }

        private JPanel buildFooter() {

            JButton today = footerButton("Today");
            today.addActionListener(e -> {
                onPick.accept(LocalDate.now());
                dispose();
            });

            JButton clear = footerButton("Clear");
            clear.addActionListener(e -> {
                onPick.accept(null);
                dispose();
            });

            JPanel footer = new JPanel(new FlowLayout(FlowLayout.CENTER, 10, 0));
            footer.setBackground(Color.WHITE);
            footer.add(today);
            footer.add(clear);

            return footer;
        }

        private JButton navButton(String text, Runnable change) {
            JButton button = new JButton(text);
            button.setFont(new Font("Arial", Font.BOLD, 16));
            button.setForeground(PURPLE);
            button.setBackground(VERY_LIGHT_PURPLE);
            button.setBorderPainted(false);
            button.setFocusPainted(false);
            button.setMargin(new Insets(0, 0, 0, 0));
            button.setPreferredSize(new Dimension(34, 30));
            button.setCursor(new Cursor(Cursor.HAND_CURSOR));

            button.addActionListener(e -> {
                change.run();
                renderMonth();
            });

            return button;
        }

        private JButton footerButton(String text) {
            JButton button = new JButton(text);
            button.setFont(new Font("Arial", Font.BOLD, 12));
            button.setForeground(PURPLE);
            button.setBackground(VERY_LIGHT_PURPLE);
            button.setBorderPainted(false);
            button.setFocusPainted(false);
            button.setPreferredSize(new Dimension(90, 30));
            button.setCursor(new Cursor(Cursor.HAND_CURSOR));
            return button;
        }

        private void renderMonth() {

            LocalDate today = LocalDate.now();

            monthLabel.setText(
                    shownMonth.getMonth().getDisplayName(
                            TextStyle.FULL, Locale.ENGLISH)
                    + " " + shownMonth.getYear());

            dayGrid.removeAll();

            for (String name : WEEKDAYS) {
                JLabel heading = new JLabel(name, SwingConstants.CENTER);
                heading.setFont(new Font("Arial", Font.BOLD, 12));
                heading.setForeground(GRAY);
                dayGrid.add(heading);
            }

            int cells = 7;

            int leadingBlanks =
                    shownMonth.atDay(1).getDayOfWeek().getValue() % 7;

            for (int i = 0; i < leadingBlanks; i++) {
                dayGrid.add(new JLabel(""));
                cells++;
            }

            for (int day = 1; day <= shownMonth.lengthOfMonth(); day++) {
                dayGrid.add(createDayButton(shownMonth.atDay(day), today));
                cells++;
            }

            // Pad to a full grid so every month has identical cell sizes.
            while (cells < GRID_ROWS * 7) {
                dayGrid.add(new JLabel(""));
                cells++;
            }

            dayGrid.revalidate();
            dayGrid.repaint();
        }

        private JButton createDayButton(LocalDate date, LocalDate today) {

            JButton button = new JButton(String.valueOf(date.getDayOfMonth()));

            button.setFont(new Font("Arial", Font.PLAIN, 13));
            button.setMargin(new Insets(0, 0, 0, 0));
            button.setHorizontalAlignment(SwingConstants.CENTER);
            button.setFocusPainted(false);
            button.setBorderPainted(false);
            button.setOpaque(true);
            button.setCursor(new Cursor(Cursor.HAND_CURSOR));

            if (date.equals(currentSelection)) {
                button.setBackground(PURPLE);
                button.setForeground(Color.WHITE);

            } else if (date.equals(today)) {
                button.setBackground(LIGHT_PURPLE);
                button.setForeground(PURPLE);

            } else {
                button.setBackground(Color.WHITE);
                button.setForeground(TEXT);
            }

            button.addActionListener(e -> {
                onPick.accept(date);
                dispose();
            });

            return button;
        }
    }
