    private void LogInActionPerformed(java.awt.event.ActionEvent evt) {                                      
        // TODO add your handling code here:
         // TODO add your handling code here:
        String user = username.getText().trim();
        String userPassword = new String(password.getPassword());
 
        if (user.isEmpty() || userPassword.isEmpty()) {
 
            JOptionPane.showMessageDialog(
                    this,
                    "Please enter your username/email and password.",
                    "Login Error",
                    JOptionPane.ERROR_MESSAGE
            );
 
            return;
        }
 
        String sql
                = "SELECT full_name, password, role, is_archived FROM users "
                + "WHERE username = ? OR email = ?";
 
        try (java.sql.Connection conn = DBConnection.connect()) {
 
            if (conn == null) {
 
                JOptionPane.showMessageDialog(
                        this,
                        "Cannot connect to the database.\n"
                        + "Make sure MySQL is running.",
                        "Connection Error",
                        JOptionPane.ERROR_MESSAGE
                );
 
                return;
            }
 
            try (java.sql.PreparedStatement pst
                    = conn.prepareStatement(sql)) {
 
                pst.setString(1, user);
                pst.setString(2, user);
 
                try (java.sql.ResultSet rs = pst.executeQuery()) {
 
                    if (rs.next()
                            && userPassword.equals(rs.getString("password"))) {
 
                        String name = rs.getString("full_name");
                        String role = rs.getString("role");
 
                        if (role == null || role.trim().isEmpty()) {
                            role = "Customer";
                        }
 
                        // Archived accounts (e.g. resigned staff) cannot log in
                        if (rs.getBoolean("is_archived")) {
 
                            JOptionPane.showMessageDialog(
                                    this,
                                    "This account has been archived.\n"
                                    + "Please contact the administrator.",
                                    "Account Archived",
                                    JOptionPane.WARNING_MESSAGE
                            );
 
                            return;
                        }
 
                        if (role.equalsIgnoreCase("Admin")
                                || role.equalsIgnoreCase("Staff")) {
 
                            // Staff / Admin -> manage all bookings
                            new AdminBookingsPage(name, role)
                                    .setVisible(true);
 
                        } else {
 
                            // Customer -> normal home page
                            new HomePagee(name).setVisible(true);
                        }
 
                        dispose();
 
                    } else {
 
                        JOptionPane.showMessageDialog(
                                this,
                                "Incorrect username/email or password.",
                                "Login Error",
                                JOptionPane.ERROR_MESSAGE
                        );
                    }
                }
            }
 
        } catch (java.sql.SQLException e) {
 
            JOptionPane.showMessageDialog(
                    this,
                    "Database error:\n" + e.getMessage(),
                    "Database Error",
                    JOptionPane.ERROR_MESSAGE
            );
 
            e.printStackTrace();
        }


    }

static class GradientBackgroundPanel extends JPanel {
 
    @Override
    protected void paintComponent(Graphics g) {
        super.paintComponent(g);
 
        Graphics2D g2 = (Graphics2D) g.create();
 
        GradientPaint gp = new GradientPaint(
                0, 0,
                new Color(240, 220, 235),
                getWidth(), getHeight(),
                new Color(210, 225, 245)
        );
 
        g2.setPaint(gp);
        g2.fillRect(0, 0, getWidth(), getHeight());
 
        g2.dispose();
    }
}

    }

    // =========================================================
    // ADD PAIR ROW  (two labelled fields side by side)
    // =========================================================
    private int addPairRow(
            JPanel panel,
            GridBagConstraints c,
            int row,
            int topGap,
            String leftLabel,
            JTextField leftField,
            String rightLabel,
            JTextField rightField) {
 
        JPanel pair = new JPanel(new GridLayout(1, 2, 15, 0));
        pair.setBackground(Color.WHITE);
        pair.add(buildNameColumn(leftLabel, leftField));
        pair.add(buildNameColumn(rightLabel, rightField));
 
        c.gridy = row++;
        c.insets = new Insets(topGap, 0, 0, 0);
        panel.add(pair, c);
 
        return row;
    }
 
    private JPanel buildNameColumn(String labelText, JTextField field) {
 
        JLabel label = new JLabel(labelText);
        label.setFont(new Font("SansSerif", Font.BOLD, 13));
        label.setForeground(Color.BLACK);
        label.setBorder(new EmptyBorder(0, 3, 0, 0));
 
        styleField(field);
        field.setPreferredSize(new Dimension(130, 35));
 
        JPanel column = new JPanel(new BorderLayout(0, 3));
        column.setBackground(Color.WHITE);
        column.add(label, BorderLayout.NORTH);
        column.add(field, BorderLayout.CENTER);
 
        return column;
    }

    // DIGITS-ONLY INPUT FILTER (used by the contact number)
    // =========================================================
    private static class DigitsOnlyFilter extends DocumentFilter {
 
        private final int maxLength;
 
        DigitsOnlyFilter(int maxLength) {
            this.maxLength = maxLength;
        }
 
        @Override
        public void insertString(
                FilterBypass fb, int offset, String text, AttributeSet attrs
        ) throws BadLocationException {
            replace(fb, offset, 0, text, attrs);
        }
 
        @Override
        public void replace(
                FilterBypass fb, int offset, int length,
                String text, AttributeSet attrs
        ) throws BadLocationException {
 
            String current = fb.getDocument().getText(
                    0, fb.getDocument().getLength());
 
            String replacement = text == null ? "" : text;
 
            String next = current.substring(0, offset)
                    + replacement
                    + current.substring(offset + length);
 
            // numbers only, and never longer than the limit
            if (next.length() <= maxLength && next.matches("[0-9]*")) {
                super.replace(fb, offset, length, replacement, attrs);
            }
        }
    }

    // REGISTER LOGIC  (new accounts are always Customers)
    // =========================================================
    private void registerUser() {
 
        String firstName = regFirstName.getText().trim();
        String lastName = regLastName.getText().trim();
        String name = firstName + " " + lastName;   // saved in full_name
        String user = regUsername.getText().trim();
        String contact = regContact.getText().trim();
        String mail = regEmail.getText().trim();
        String pass = new String(regPassword.getPassword());
        String confirm = new String(regConfirm.getPassword());
 
        // VALIDATION
        if (firstName.isEmpty() || lastName.isEmpty() || user.isEmpty()
                || contact.isEmpty() || mail.isEmpty()
                || pass.isEmpty() || confirm.isEmpty()) {
            showRegError("Please fill in all fields.");
            return;
        }
 
        if (firstName.length() < 2 || lastName.length() < 2) {
            showRegError(
                    "First and last name must be at least 2 characters each."
            );
            return;
        }
 
        if (user.length() < 4) {
            showRegError("Username must be at least 4 characters.");
            return;
        }
 
        if (!user.matches("^[a-zA-Z0-9_]+$")) {
            showRegError(
                    "Username can only contain letters, numbers, and underscore."
            );
            return;
        }
 
        if (!contact.matches("^[0-9]{" + CONTACT_LENGTH + "}$")) {
            showRegError(
                    "Contact number must be exactly " + CONTACT_LENGTH
                    + " digits (numbers only).\n"
                    + "Example: 09123456789"
            );
            regContact.requestFocusInWindow();
            return;
        }
 
        if (!mail.matches("^[\\w.+-]+@[\\w-]+(\\.[\\w-]+)+$")) {
            showRegError("Please enter a valid email address.");
            return;
        }
 
        if (pass.length() < 6) {
            showRegError("Password must be at least 6 characters.");
            return;
        }
 
        if (!pass.equals(confirm)) {
            showRegError("Passwords do not match.");
            return;
        }
 
        // DATABASE
        String checkSql
                = "SELECT 1 FROM users "
                + "WHERE username = ? OR email = ?";
 
        // role is not set here: the database default is 'Customer'
        String insertSql
                = "INSERT INTO users "
                + "(full_name, username, email, contact_number, password) "
                + "VALUES (?, ?, ?, ?, ?)";
 
        try (java.sql.Connection conn = DBConnection.connect()) {
 
            if (conn == null) {
 
                JOptionPane.showMessageDialog(
                        this,
                        "Cannot connect to the database.\n"
                        + "Make sure MySQL and XAMPP are running.",
                        "Connection Error",
                        JOptionPane.ERROR_MESSAGE
                );
 
                return;
            }
 
            // CHECK DUPLICATE USERNAME / EMAIL
            try (java.sql.PreparedStatement check
                    = conn.prepareStatement(checkSql)) {
 
                check.setString(1, user);
                check.setString(2, mail);
 
                try (java.sql.ResultSet rs = check.executeQuery()) {
 
                    if (rs.next()) {
                        showRegError(
                                "That username or email is already registered."
                        );
                        return;
                    }
                }
            }
 
            // INSERT ACCOUNT
            try (java.sql.PreparedStatement ins
                    = conn.prepareStatement(insertSql)) {
 
                ins.setString(1, name);
                ins.setString(2, user);
                ins.setString(3, mail);
                ins.setString(4, contact);
                ins.setString(5, pass);
 
                int result = ins.executeUpdate();
 
                if (result > 0) {
 
                    JOptionPane.showMessageDialog(
                            this,
                            "Account created successfully!\n"
                            + "You can now log in.",
                            "Registration Successful",
                            JOptionPane.INFORMATION_MESSAGE
                    );
 
                    clearRegisterForm();
 
                    username.setText(user);
                    password.setText("");
 
                    showCard("login");
 
                    password.requestFocusInWindow();
                }
            }
 
        } catch (java.sql.SQLException e) {
 
            JOptionPane.showMessageDialog(
                    this,
                    "Database error:\n" + e.getMessage(),
                    "Database Error",
                    JOptionPane.ERROR_MESSAGE
            );
 
            e.printStackTrace();
        }
    }

    static class CircularBadgePanel extends JPanel {
 
    @Override
    protected void paintComponent(Graphics g) {
        super.paintComponent(g);
 
        Graphics2D g2 = (Graphics2D) g.create();
 
        g2.setRenderingHint(
                RenderingHints.KEY_ANTIALIASING,
                RenderingHints.VALUE_ANTIALIAS_ON
        );
 
        int w = getWidth();
        int h = getHeight();
        int size = Math.min(w, h) - 8;
 
        float x = (w - size) / 2f;
        float y = (h - size) / 2f;
        float cx = w / 2f;
 
        // LAVENDER CIRCLE
        g2.setColor(new Color(174, 137, 195));
        g2.fill(new Ellipse2D.Float(x, y, size, size));
 
        // WHITE CIRCULAR BORDER
        g2.setColor(Color.WHITE);
        g2.setStroke(new BasicStroke(2f));
 
        float ring = size * 0.025f;
 
        g2.draw(new Ellipse2D.Float(
                x + ring, y + ring,
                size - ring * 2,
                size - ring * 2
        ));
 
        // WHITE WATER DROPLET
        g2.setStroke(new BasicStroke(
                Math.max(1.8f, size * 0.009f),
                BasicStroke.CAP_ROUND,
                BasicStroke.JOIN_ROUND
        ));
 
        Path2D.Float drop = new Path2D.Float();
 
        float top = y + size * 0.22f;
        float bottom = y + size * 0.46f;
 
        drop.moveTo(cx, top);
 
        drop.curveTo(
                cx - size * 0.09f, y + size * 0.33f,
                cx - size * 0.105f, y + size * 0.39f,
                cx - size * 0.055f, y + size * 0.44f
        );
 
        drop.curveTo(
                cx - size * 0.025f, bottom + size * 0.015f,
                cx + size * 0.025f, bottom + size * 0.015f,
                cx + size * 0.055f, y + size * 0.44f
        );
 
        drop.curveTo(
                cx + size * 0.105f, y + size * 0.39f,
                cx + size * 0.09f, y + size * 0.33f,
                cx, top
        );
 
        drop.closePath();
        g2.draw(drop);
 
        // LAVANDERA TEXT
        g2.setColor(Color.WHITE);
 
        String brand = "Lavandera";
 
        g2.setFont(new Font(
                "Serif",
                Font.BOLD,
                Math.max(12, Math.round(size * 0.115f))
        ));
 
        FontMetrics fm = g2.getFontMetrics();
 
        g2.drawString(
                brand,
                cx - fm.stringWidth(brand) / 2f,
                y + size * 0.61f
        );
 
        // LAUNDRY SHOP TEXT
        String subtitle = "Laundry Shop";
 
        g2.setFont(new Font(
                "SansSerif",
                Font.PLAIN,
                Math.max(9, Math.round(size * 0.059f))
        ));
 
        fm = g2.getFontMetrics();
 
        g2.drawString(
                subtitle,
                cx - fm.stringWidth(subtitle) / 2f,
                y + size * 0.69f
        );
 
        g2.dispose();
    }
