    private JButton createSidebarButton(String text) {

        JButton button = new JButton(text);
        button.setFont(new Font("Arial", Font.BOLD, 12));
        button.setForeground(TEXT);
        button.setBackground(Color.WHITE);
        button.setHorizontalAlignment(SwingConstants.LEFT);
        button.setBorder(BorderFactory.createEmptyBorder(0, 15, 0, 0));
        button.setBorderPainted(false);
        button.setFocusPainted(false);
        button.setCursor(new Cursor(Cursor.HAND_CURSOR));

        button.addMouseListener(new MouseAdapter() {
            @Override
            public void mouseEntered(MouseEvent e) {
                if (button != activeButton) {
                    button.setBackground(VERY_LIGHT_PURPLE);
                    button.setForeground(PURPLE);
                }
            }

            @Override
            public void mouseExited(MouseEvent e) {
                applySidebarStyle(button);
            }
        });

        return button;
    }

    /** Highlights the button of the current view, resets the others. */
    private void applySidebarStyle(JButton button) {

        boolean selected = button == activeButton;

        button.setForeground(selected ? PURPLE : TEXT);
        button.setBackground(selected ? LIGHT_PURPLE : Color.WHITE);
    }

    /**
     * Switches between the Home view and the About Us view.

        applySidebarStyle(homeButton);
        applySidebarStyle(aboutButton);

        }
    }

    /**
     * The window can be resized, and ResponsiveUI scales every component
     * from the bounds it was DESIGNED with. These helpers therefore move a
     * component in design coordinates and let the layout scale it. (Moving
     * it with setLocation() made the cards grow and overlap whenever the
     * window was larger than 1200x750.)
     */
    private Rectangle designBounds(JComponent c) {

        Object stored = c.getClientProperty("designBounds");

        if (stored instanceof Rectangle) {
            return (Rectangle) stored;
        }

        Rectangle b = c.getBounds();
        double sx = 1.0;
        double sy = 1.0;

        // Before the window is shown the bounds are still the design ones.
        if (isShowing() && getContentPane().getWidth() > 0) {
            sx = getContentPane().getWidth() / 1200.0;
            sy = getContentPane().getHeight() / 750.0;
        }

        Rectangle design = new Rectangle(
                b.x, b.y,
                (int) Math.round(b.width / sx),
                (int) Math.round(b.height / sy)
        );

        c.putClientProperty("designBounds", design);
        return design;
    }

    private void moveInDesign(JComponent c, Rectangle design, int x, int y) {

        c.setBounds(x, y, design.width, design.height);

        Container parent = c.getParent();

        if (parent != null) {
            parent.revalidate();
        }
    }

    private void animateSlide(
            JComponent component,
            int finalX, int finalY,
            int startX, int startY,
            int delay
    ) {
        final Rectangle design = designBounds(component);

        moveInDesign(component, design, startX, startY);

        Timer timer = new Timer(15, null);
        timer.setInitialDelay(delay);

        final long[] startTime = {0};

        timer.addActionListener(e -> {

            if (startTime[0] == 0) {
                startTime[0] = System.currentTimeMillis();
            }

            double progress =
                    (System.currentTimeMillis() - startTime[0]) / 450.0;

            if (progress >= 1) {
                moveInDesign(component, design, finalX, finalY);
                timer.stop();
                return;
            }

            double eased = 1 - Math.pow(1 - progress, 3);

            moveInDesign(
                    component, design,
                    (int) (startX + (finalX - startX) * eased),
                    (int) (startY + (finalY - startY) * eased)
            );
        });

        timer.start();
    }

    private void animateSidebar(JComponent component, int finalY, int delay) {

        final Rectangle design = designBounds(component);
        final int finalX = design.x;

        moveInDesign(component, design, -230, finalY);

        Timer timer = new Timer(15, null);
        timer.setInitialDelay(delay);

        final long[] startTime = {0};

        timer.addActionListener(e -> {

            if (startTime[0] == 0) {
                startTime[0] = System.currentTimeMillis();
            }

            double progress =
                    (System.currentTimeMillis() - startTime[0]) / 400.0;

            if (progress >= 1) {
                moveInDesign(component, design, finalX, finalY);
                timer.stop();
                return;
            }

            double eased = 1 - Math.pow(1 - progress, 3);

            moveInDesign(
                    component, design,
                    (int) (-230 + (finalX + 230) * eased),
                    finalY
            );
        });

        timer.start();
    }

    /** Small drawn icon (no font needed, so it never shows as an empty box). */
    private static class CardIcon extends JPanel {

        static final int BOOK = 0;
        static final int LIST = 1;

        private final int type;

        CardIcon(int type) {
            this.type = type;
            setOpaque(false);
        }

        @Override
        protected void paintComponent(Graphics g) {

            Graphics2D g2 = (Graphics2D) g.create();

            g2.setRenderingHint(
                    RenderingHints.KEY_ANTIALIASING,
                    RenderingHints.VALUE_ANTIALIAS_ON
            );

            int w = getWidth();
            int h = getHeight();

            g2.setColor(LIGHT_PURPLE);
            g2.fillRoundRect(0, 0, w, h, 14, 14);

            g2.setColor(PURPLE);
            g2.setStroke(new BasicStroke(
                    Math.max(2f, w / 16f),
                    BasicStroke.CAP_ROUND,
                    BasicStroke.JOIN_ROUND
            ));

            if (type == BOOK) {
                // calendar with a plus
                int x = w * 26 / 100;
                int y = h * 30 / 100;
                int cw = w * 48 / 100;
                int ch = h * 44 / 100;

                g2.drawRoundRect(x, y, cw, ch, 6, 6);
                g2.drawLine(x, y + ch * 28 / 100, x + cw, y + ch * 28 / 100);
                g2.drawLine(x + cw * 28 / 100, y - h * 6 / 100,
                        x + cw * 28 / 100, y + h * 5 / 100);
                g2.drawLine(x + cw * 72 / 100, y - h * 6 / 100,
                        x + cw * 72 / 100, y + h * 5 / 100);

                int cx = x + cw / 2;
                int cy = y + ch * 64 / 100;
                int r = Math.max(3, cw / 8);
                g2.drawLine(cx - r, cy, cx + r, cy);
                g2.drawLine(cx, cy - r, cx, cy + r);

            } else {
                // list with three rows
                for (int i = 0; i < 3; i++) {
                    int ly = h * 32 / 100 + i * h * 18 / 100;
                    int dot = Math.max(3, w / 14);

                    g2.fillOval(w * 28 / 100 - dot / 2, ly - dot / 2, dot, dot);
                    g2.drawLine(w * 42 / 100, ly, w * 74 / 100, ly);
                }
            }

            g2.dispose();
        }
    }

    private static class RoundedPanel extends JPanel {

        private final Color color;
        private final int radius;

        RoundedPanel(Color color, int radius) {
            this.color = color;
            this.radius = radius;
            setOpaque(false);
        }

        @Override
        protected void paintComponent(Graphics g) {

            Graphics2D g2 = (Graphics2D) g.create();

            g2.setRenderingHint(
                    RenderingHints.KEY_ANTIALIASING,
                    RenderingHints.VALUE_ANTIALIAS_ON
            );

            g2.setColor(color);
            g2.fillRoundRect(0, 0, getWidth(), getHeight(), radius, radius);
            g2.dispose();

            super.paintComponent(g);
        }
    }

        }
    }

            // Inner circle
            g2.setColor(new Color(175, 140, 200));

            // Inner white border
            g2.setColor(Color.WHITE);
            g2.setStroke(new BasicStroke(2));
            g2.drawOval(22, 22, 166, 126);

            // Drop
            g2.setColor(Color.WHITE);

            Path2D drop = new Path2D.Double();
            drop.moveTo(105, 35);
            drop.curveTo(85, 55, 78, 65, 78, 75);
