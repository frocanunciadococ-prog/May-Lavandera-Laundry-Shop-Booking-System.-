    private final Color LIGHT_PURPLE = new Color(220, 205, 235);

    private final Color TEXT = new Color(55, 45, 65);
    private final Color GRAY = new Color(110, 105, 115);

    private JPanel bookingsPanel;

        setResizable(true);

        JPanel main = new JPanel(null);
        main.setBackground(new Color(248, 245, 250));
        setContentPane(main);

        // =====================================================
        // TITLE
        // =====================================================

        JLabel title = new JLabel("My Bookings");
        title.setFont(new Font("Arial", Font.BOLD, 24));
        title.setForeground(TEXT);
        title.setBounds(50, 35, 400, 40);
        main.add(title);

        JLabel userLabel = new JLabel(
                "Bookings for: " + userName
        );
        userLabel.setFont(new Font("Arial", Font.PLAIN, 13));
        userLabel.setForeground(GRAY);
        userLabel.setBounds(52, 82, 500, 25);
        main.add(userLabel);

        // =====================================================
        // BACK
        // =====================================================

        JButton backButton = new JButton("← Back");
        backButton.setFont(new Font("Arial", Font.BOLD, 12));
        backButton.setForeground(PURPLE);
        backButton.setBackground(VERY_LIGHT_PURPLE);
        backButton.setBorderPainted(false);
        backButton.setFocusPainted(false);
        backButton.setBounds(1000, 45, 100, 35);
        main.add(backButton);

        // =====================================================
        // SCROLL
        // =====================================================

        bookingsPanel = new JPanel(null);
        bookingsPanel.setBackground(new Color(248, 245, 250));

        JScrollPane scrollPane = new JScrollPane(bookingsPanel);
        scrollPane.setBounds(50, 130, 1050, 520);
        scrollPane.setBorder(BorderFactory.createEmptyBorder());
        scrollPane.getVerticalScrollBar().setUnitIncrement(16);

        // =====================================================
        // ACTION
        // =====================================================

        backButton.addActionListener(e -> {
            new HomePagee(userName).setVisible(true);
            dispose();
        });

        // =====================================================
        // LOAD BOOKINGS
        // =====================================================

        loadBookings();

        // =====================================================
        // ANIMATION
        // =====================================================

        animateSlide(title, 50, 35, 50, -50, 0);
        animateSlide(userLabel, 52, 82, -300, 82, 100);
        animateSlide(backButton, 1000, 45, 1200, 45, 150);

        addHover(
                backButton,
                VERY_LIGHT_PURPLE,
                LIGHT_PURPLE
        );

        ResponsiveUI.enable(this, 1200, 750);
    }

    // =====================================================
    // LOAD BOOKINGS
    // =====================================================

    private void loadBookings() {

        bookingsPanel.removeAll();

        String sql =
                "SELECT booking_id, service, booking_date, "
                + "booking_time, booking_status, payment_status, "
                + "quantity, amount "
                + "FROM tbl_bookings "
                + "WHERE username = ? "
                + "ORDER BY booking_id DESC";

        int y = 20;
        int animationDelay = 100;
        int count = 0;

        try (
                Connection conn = DBConnection.connect();
                PreparedStatement pst = conn.prepareStatement(sql)
        ) {

            if (conn == null) {
                showNoBookings("Database connection failed.");
                return;
            }

            pst.setString(1, userName);

            ResultSet rs = pst.executeQuery();

            while (rs.next()) {

                int bookingId =
                        rs.getInt("booking_id");

                String service =
                        rs.getString("service");

                String date =
                        rs.getString("booking_date");

                Time sqlTime =
                        rs.getTime("booking_time");

                String bookingTime = "";

                if (sqlTime != null) {
                    bookingTime = sqlTime
                            .toLocalTime()
                            .format(
                                    DateTimeFormatter.ofPattern("h:mma")
                            );
                }

                String bookingStatus =
                        rs.getString("booking_status");

                String paymentStatus =
                        rs.getString("payment_status");

                double quantity =
                        rs.getDouble("quantity");

                double amount =
                        rs.getDouble("amount");

                JPanel card = createBookingCard(
                        bookingId,
                        service,
                        quantity,
                        date,
                        bookingTime,
                        amount,
                        bookingStatus,
                        paymentStatus
                );

                card.setBounds(
                        20,
                        y,
                        980,
                        180
                );

                bookingsPanel.add(card);

                animateSlide(
                        card,
                        20,
                        y,
                        1050,
                        y,
                        animationDelay
                );

                y += 195;
                animationDelay += 100;
                count++;
            }

            if (count == 0) {
                showNoBookings(
                        "You don't have any bookings yet."
                );
            } else {
                bookingsPanel.setPreferredSize(
                        new Dimension(1020, y + 20)
                );
            }

            bookingsPanel.revalidate();
            bookingsPanel.repaint();

        } catch (SQLException e) {

            showNoBookings(
                    "Unable to load bookings.\n"
                    + e.getMessage()
            );
        }
    }

        // =====================================================

                cancelButton,

                new Color(125, 75, 165)
        );

        return card;
    }

    // =====================================================
    // CANCEL BOOKING
    // =====================================================

    private void cancelBooking(int bookingId) {

        String sql =
                "UPDATE tbl_bookings "
                + "SET booking_status = 'Cancelled' "
                + "WHERE booking_id = ? "
                + "AND username = ?";

        try (
                Connection conn = DBConnection.connect();
                PreparedStatement pst = conn.prepareStatement(sql)
        ) {

            if (conn == null) {

                JOptionPane.showMessageDialog(
                        this,
                        "Database connection failed."
                );

                return;
            }

            pst.setInt(1, bookingId);
            pst.setString(2, userName);

            pst.executeUpdate();

            JOptionPane.showMessageDialog(
                    this,
                    "Booking cancelled."
            );

            loadBookings();

        } catch (SQLException e) {

            JOptionPane.showMessageDialog(
                    this,
                    "Database error:\n" + e.getMessage()
            );
        }
    }

    // =====================================================
    // NO BOOKINGS
    // =====================================================

    private void showNoBookings(String message) {

        bookingsPanel.removeAll();

        JLabel label = new JLabel(
                "<html><center>"
                + message.replace("\n", "<br>")
                + "</center></html>",
                SwingConstants.CENTER
        );

        label.setFont(
                new Font("Arial", Font.PLAIN, 13)
        );
        label.setForeground(GRAY);
        label.setBounds(200, 170, 600, 80);

        bookingsPanel.add(label);

        bookingsPanel.setPreferredSize(
                new Dimension(1020, 500)
        );

        bookingsPanel.revalidate();
        bookingsPanel.repaint();
    }

    // =====================================================
    // HOVER
    // =====================================================

    private void addHover(
            JButton button,
            Color normal,
            Color hover
    ) {

        button.addMouseListener(
                new MouseAdapter() {

                    @Override
                    public void mouseEntered(
                            MouseEvent e
                    ) {
                        button.setBackground(hover);
                    }

                    @Override
                    public void mouseExited(
                            MouseEvent e
                    ) {
                        button.setBackground(normal);
                    }
                }
        );
    }

    // =====================================================
    // ANIMATION
    // =====================================================

    private void animateSlide(
            JComponent component,
            int finalX,
            int finalY,
            int startX,
            int startY,
            int delay
    ) {

        component.setLocation(
                startX,
                startY
        );

        Timer timer = new Timer(15, null);
        timer.setInitialDelay(delay);

        final long[] startTime = {0};

        timer.addActionListener(e -> {

            if (startTime[0] == 0) {
                startTime[0] =
                        System.currentTimeMillis();
            }

            double progress =
                    (
                            System.currentTimeMillis()
                            - startTime[0]
                    ) / 450.0;

            if (progress >= 1) {

                component.setLocation(
                        finalX,
                        finalY
                );

                timer.stop();
                return;
            }

            double eased =
                    1 - Math.pow(
                            1 - progress,
                            3
                    );

            int x =
                    (int) (
                            startX
                            + (
                                    finalX
                                    - startX
                            ) * eased
                    );

            int y =
                    (int) (
                            startY
                            + (
                                    finalY
                                    - startY
                            ) * eased
                    );

            component.setLocation(x, y);
        });

        timer.start();
    }
}
